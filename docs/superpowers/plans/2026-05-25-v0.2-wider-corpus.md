# ato-mcp v0.2 — wider corpus implementation plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task.

**Goal:** Implement the v0.2 wider-corpus spec — add legislation, ATO legal database, state revenue, point-in-time queries, new tools (`get_definition`, `get_doc`, `get_doc_anchors`, `get_threshold`), citation/anchor graph, and GitHub-release-based corpus updates.

**Architecture:** v0.1 monorepo extended. Pipeline gains a `sources/` sub-package (each source is a Python module emitting unified `Doc` + `Chunk`). MCP server gains 4 new tools + `pit` plumbing. Schema is additive — v0.1 clients keep working against v0.2 corpora.

**Tech Stack:** Same as v0.1. Adds `httpx[http2]` for legislation.gov.au XML, `lxml` for fast XML parsing, `python-wordnet` (or `wn`) for ordinary-meaning fallback, `tenacity` for retry, `pyzstd` for corpus compression.

---

## Execution priority (autonomous overnight run)

Tasks are organised into 5 phases. **Execute in this order**, committing after each phase. If time runs short, ship what's complete and document remaining work in HANDOFF.

- **Phase A** — Framework: schema migration, new tool stubs, PIT plumbing, doc_type taxonomy expansion. Doesn't add content but unblocks all later phases.
- **Phase B** — Legislation source ingest + statutory definitions + thresholds. Highest content value per hour of work.
- **Phase C** — law.ato.gov.au public rulings only (~5k docs). PBRs and case summaries deferred to v0.2-c2 if time allows.
- **Phase D** — Re-classify existing ato.gov.au corpus by doc_type + rebuild corpus with new sources mixed in.
- **Phase E** — GitHub release flow + HANDOFF update.

State revenue (8 jurisdictions) and edited PBRs (~120k docs) are deliberately deferred to follow-up plans `v0.2.5` and `v0.2.6` — too much work for one autonomous session.

```ts
async keywordSearch(query: string, k: number, pit?: string): Promise<SearchHit[]> {
  const sanitised = query.replace(/[^\p{L}\p{N}\s]/gu, " ").trim();
  if (!sanitised) return [];
  const pitClause = pit
    ? "AND (c.effective_from IS NULL OR c.effective_from <= ?) AND (c.effective_to IS NULL OR c.effective_to > ?)"
    : "";
  const sql = `
    SELECT c.chunk_id, c.doc_id, c.ord, c.text, c.heading_path,
           d.title, d.url, d.doc_type,
           bm25(fts_chunks) AS bm25_score
    FROM fts_chunks
    JOIN chunks c ON c.chunk_id = fts_chunks.chunk_id
    JOIN docs   d ON d.doc_id  = c.doc_id
    WHERE fts_chunks MATCH ? ${pitClause}
    ORDER BY bm25_score
    LIMIT ?`;
  const params: any[] = [sanitised];
  if (pit) params.push(pit, pit);
  params.push(k);
  const rows = this.db.prepare(sql).all(...params) as Array<ChunkRow & { bm25_score: number }>;
  return rows.map((r, i) => this.rowToHit(r, 1.0 / (1 + i)));
}

// Mirror pit filter into vectorSearch and getChunks similarly.

async getDoc(docId: string) {
  const doc = this.db.prepare("SELECT * FROM docs WHERE doc_id = ?").get(docId) as any;
  if (!doc) return null;
  const anchors = this.db.prepare("SELECT * FROM anchors WHERE doc_id = ?").all(docId) as AnchorRow[];
  return { doc, cleaned_html: null, anchors };  // cleaned_html populated in v0.2-c
}

async getDocAnchors(docId: string) {
  const anchors = this.db.prepare("SELECT * FROM anchors WHERE doc_id = ?").all(docId) as AnchorRow[];
  const inbound = this.db.prepare("SELECT * FROM citations WHERE to_doc_id = ?").all(docId) as CitationRow[];
  const outbound = this.db.prepare(`
    SELECT cc.* FROM citations cc
    JOIN chunks ck ON ck.chunk_id = cc.from_chunk_id
    WHERE ck.doc_id = ?`).all(docId) as CitationRow[];
  return { anchors, inbound, outbound };
}

async getDefinition(term: string, pit: string | null): Promise<DefinitionRow[]> {
  const pitClause = pit
    ? "AND (effective_from IS NULL OR effective_from <= ?) AND (effective_to IS NULL OR effective_to > ?)"
    : "";
  const sql = `SELECT * FROM definitions WHERE term = ? COLLATE NOCASE ${pitClause}`;
  const params: any[] = [term];
  if (pit) params.push(pit, pit);
  return this.db.prepare(sql).all(...params) as DefinitionRow[];
}

async getThreshold(name: string, pit: string | null): Promise<ThresholdRow | null> {
  const pitClause = pit
    ? "AND (effective_from IS NULL OR effective_from <= ?) AND (effective_to IS NULL OR effective_to > ?)"
    : "";
  const sql = `SELECT * FROM thresholds WHERE name = ? ${pitClause} ORDER BY effective_from DESC LIMIT 1`;
  const params: any[] = [name];
  if (pit) params.push(pit, pit);
  return (this.db.prepare(sql).get(...params) as ThresholdRow | undefined) ?? null;
}
```

```ts
import type { GetDefinitionInput } from "@ato-mcp/shared";
import type { Store } from "../store/types.js";

export interface GetDefinitionDeps {
  store: Store | null;
  wordnetLookup?: (term: string) => Promise<string | null>;
}

export interface GetDefinitionOutput {
  term: string;
  kind: "statutory" | "ordinary";
  source?: { doc_id: string; anchor: string | null; citation: string };
  body: string;
  effective_from?: string | null;
  effective_to?: string | null;
  alternatives?: Array<{ doc_id: string; citation: string; body: string }>;
}

export async function getDefinition(
  deps: GetDefinitionDeps,
  args: GetDefinitionInput,
): Promise<GetDefinitionOutput> {
  if (!deps.store) {
    throw new Error("Corpus not installed. Run `ato-mcp update`.");
  }
  const pit = args.pit ?? new Date().toISOString().slice(0, 10);
  const matches = await deps.store.getDefinition(args.term, pit);
  if (matches.length > 0) {
    const primary = matches[0]!;
    return {
      term: args.term,
      kind: "statutory",
      source: { doc_id: primary.doc_id, anchor: primary.anchor_id, citation: primary.doc_id },
      body: primary.body,
      effective_from: primary.effective_from,
      effective_to: primary.effective_to,
      alternatives: matches.slice(1).map(m => ({ doc_id: m.doc_id, citation: m.doc_id, body: m.body })),
    };
  }
  // Ordinary-meaning fallback
  if (deps.wordnetLookup) {
    const body = await deps.wordnetLookup(args.term);
    if (body) {
      return { term: args.term, kind: "ordinary", body: `${body}\n\n(Source: Open English WordNet 2024, CC-BY 4.0. Not a statutory definition.)` };
    }
  }
  return { term: args.term, kind: "ordinary", body: `No statutory definition found for "${args.term}".` };
}
```

```ts
export async function fetchUri(args: FetchInput): Promise<FetchOutput> {
  const uri = args.uri.trim();
  if (uri.startsWith("ato:")) {
    const tail = uri.slice("ato:".length).replace(/^\/+/, "");
    return await fetchHttp(uri, `https://www.ato.gov.au/${tail}`);
  }
  if (uri.startsWith("ato-law:")) {
    const tail = uri.slice("ato-law:".length).replace(/^\/+/, "");
    return await fetchHttp(uri, `https://www.ato.gov.au/law/view.htm?docid=${tail}`);
  }
  if (uri.startsWith("legis:")) {
    const [actKey, section] = uri.slice("legis:".length).split("/");
    return await fetchHttp(uri, `https://www.legislation.gov.au/Latest/${actKey}/${section}`);
  }
  if (uri.startsWith("staterev-")) {
    const m = uri.match(/^staterev-([a-z]+):(.+)$/);
    if (m) {
      const [, juris, path] = m;
      const host = stateRevenueHost(juris!);
      if (host) return await fetchHttp(uri, `${host}/${path}`);
    }
  }
  throw new Error(`Unsupported URI scheme: ${uri}`);
}

function stateRevenueHost(juris: string): string | null {
  const hosts: Record<string, string> = {
    nsw: "https://www.revenue.nsw.gov.au",
    vic: "https://www.sro.vic.gov.au",
    qld: "https://qro.qld.gov.au",
    sa:  "https://www.revenuesa.sa.gov.au",
    wa:  "https://www.wa.gov.au/organisation/department-of-finance/state-revenue",
    tas: "https://www.sro.tas.gov.au",
    act: "https://www.revenue.act.gov.au",
    nt:  "https://nt.gov.au/employ/for-employers-in-nt/territory-revenue-office",
  };
  return hosts[juris] ?? null;
}
```

```python
class LegislationSource(Source):
    name = "legislation"

    def __init__(self, config: PipelineConfig, acts: list[str] | None = None):
        self.config = config
        self.acts = acts or list(ACT_CONFIG.keys())

    async def fetch(self) -> SourceOutput:
        out = SourceOutput(docs=[], chunks=[])
        async with httpx.AsyncClient(timeout=self.config.request_timeout_s, headers={"User-Agent": self.config.user_agent}) as client:
            for act_key in self.acts:
                act_cfg = ACT_CONFIG[act_key]
                latest_xml_url = await self._latest_compilation_url(client, act_cfg)
                if not latest_xml_url:
                    continue
                xml_bytes = await self._fetch_xml(client, latest_xml_url)
                self._parse_act_xml(xml_bytes, act_cfg, out)
        return out

    async def _latest_compilation_url(self, client: httpx.AsyncClient, act_cfg: dict) -> str | None:
        # The Federal Register publishes per-series compilation lists. The actual API for the
        # "latest" XML compilation lives at a stable URL using the series id; verify pattern.
        # If pattern doesn't work, fall back to scraping the HTML compilation list.
        url = f"https://www.legislation.gov.au/{act_cfg['series_id']}/latest/text/xml"
        try:
            resp = await client.head(url, follow_redirects=True)
            if resp.status_code == 200:
                return url
        except httpx.HTTPError:
            pass
        return None  # implementer note: real implementation may need HTML parse fallback

    async def _fetch_xml(self, client: httpx.AsyncClient, url: str) -> bytes:
        resp = await client.get(url, follow_redirects=True)
        resp.raise_for_status()
        return resp.content

    def _parse_act_xml(self, xml_bytes: bytes, act_cfg: dict, out: SourceOutput) -> None:
        tree = etree.fromstring(xml_bytes)
        # Federal Register XML uses standard namespacing. Strip if present for simpler XPath.
        for elem in tree.iter():
            if isinstance(elem.tag, str) and '}' in elem.tag:
                elem.tag = elem.tag.split('}', 1)[1]
        now = datetime.now(timezone.utc).isoformat()
        # Walk Chapters > Parts > Divisions > Subdivisions > Sections
        for section in tree.iter("Section"):
            section_num = (section.findtext("SectionNumber") or "").strip()
            section_heading = (section.findtext("SectionHeading") or "").strip()
            if not section_num:
                continue
            doc_id = f"legis:{act_cfg['series_id'].lower()}/{section_num.replace(' ', '')}"
            heading_path = self._heading_path_for(section)
            body_parts = []
            for p in section.iter():
                if p.tag in ("Subsection", "Paragraph", "Text"):
                    text = (p.text or "").strip()
                    if text:
                        body_parts.append(text)
            body = "\n\n".join(body_parts)[:8000]
            if not body:
                continue
            out.docs.append(Doc(
                doc_id=doc_id,
                source="legislation",
                url=f"https://www.legislation.gov.au/{act_cfg['series_id']}/latest/text",
                title=f"{act_cfg['title']} — Section {section_num} — {section_heading}",
                jurisdiction="AU",
                doc_type=act_cfg["doc_type"],
                retrieved_at=now,
                metadata={"section": section_num, "act_key": act_cfg["series_id"]},
            ))
            out.chunks.append(Chunk(
                chunk_id=f"{doc_id}#0",
                doc_id=doc_id,
                ord=0,
                text=body,
                heading_path=heading_path,
                char_start=0,
                char_end=len(body),
            ))
        # Definitions section — ITAA 1997 s 995-1 and similar. Pull each defined term.
        for definition in tree.iter("Definition"):
            term = (definition.findtext("Term") or definition.findtext("DefinedTerm") or "").strip().lower()
            body = " ".join((definition.itertext())).strip()
            if not term or not body:
                continue
            out.definitions.append({
                "term": term,
                "doc_id": f"legis:{act_cfg['series_id'].lower()}/dictionary",
                "anchor_id": None,
                "body": body[:4000],
                "effective_from": None,
                "effective_to": None,
            })

    def _heading_path_for(self, section_elem) -> list[str]:
        parts = []
        for ancestor in reversed(list(section_elem.iterancestors())):
            tag = ancestor.tag
            if tag in ("Chapter", "Part", "Division", "Subdivision"):
                head = (ancestor.findtext(f"{tag}Heading") or "").strip()
                if head:
                    parts.append(f"{tag} {head}")
        return parts
```

```python
from __future__ import annotations
import re
from dataclasses import dataclass
from typing import Callable

import httpx


@dataclass
class ThresholdExtractor:
    name: str
    url: str
    pattern: str            # regex with one capture group for the numeric value
    unit: str
    effective_from: str | None
    description: str

    def extract(self, html: str) -> float | None:
        m = re.search(self.pattern, html)
        if not m:
            return None
        raw = m.group(1).replace(",", "").replace("$", "").strip()
        try:
            return float(raw)
        except ValueError:
            return None


EXTRACTORS: list[ThresholdExtractor] = [
    ThresholdExtractor(
```

```ts
import fs from "node:fs";
import path from "node:path";
import { dataDir } from "./paths.js";

interface WordNetData {
  [term: string]: { definitions: string[] };
}

let cache: WordNetData | null = null;

export async function wordnetLookup(term: string): Promise<string | null> {
  if (!cache) cache = await loadWordnet();
  const entry = cache[term.toLowerCase()];
  if (!entry || entry.definitions.length === 0) return null;
  return entry.definitions[0]!;
}

async function loadWordnet(): Promise<WordNetData> {
  const localPath = path.join(dataDir(), "wordnet.json");
  if (!fs.existsSync(localPath)) {
    // First-run download — simple JSON dump of OEWN 2024
    // For v0.2 ship a minimal English noun subset (~5MB)
    await downloadWordnetDump(localPath);
  }
  return JSON.parse(fs.readFileSync(localPath, "utf-8"));
}

  const url = "https://en-word.net/static/oewn-noun-defs.json";  // PLACEHOLDER: real OEWN download

  const resp = await fetch(url);
  if (!resp.ok) throw new Error(`Failed to download WordNet: ${resp.status}`);
  fs.writeFileSync(target, await resp.text());
}
```

```python
class LawAtoSource(Source):
    name = "law_ato"

    TYPES = [
        ("TXR", "ATO_RULING_TR"),
        ("TXD", "ATO_RULING_TD"),
        ("GSTR", "ATO_RULING_GSTR"),
        ("GSTD", "ATO_RULING_GSTD"),
        ("PR", "ATO_RULING_PR"),
        ("CR", "ATO_RULING_CR"),
        ("LCR", "ATO_RULING_LCR"),
        ("PCG", "ATO_RULING_PCG"),
        ("MT", "ATO_RULING_MT"),
        ("FTR", "ATO_RULING_FTR"),
    ]

    def __init__(self, config: PipelineConfig, types: list[str] | None = None, max_per_type: int = 1000):
        self.config = config
        self.types = types or [t[0] for t in self.TYPES]
        self.max_per_type = max_per_type

    async def fetch(self) -> SourceOutput:
        out = SourceOutput(docs=[], chunks=[])
        async with httpx.AsyncClient(
            timeout=self.config.request_timeout_s,
            headers={"User-Agent": self.config.user_agent},
            http2=True,
        ) as client:
            for type_code, doc_type in self.TYPES:
                if type_code not in self.types:
                    continue
                urls = await self._list_rulings(client, type_code)
                for url in urls[: self.max_per_type]:
                    html = await self._fetch_page(client, url)
                    if not html:
                        continue
                    self._parse_ruling(html, url, type_code, doc_type, out)
                    await asyncio.sleep(self.config.request_per_host_delay_s)
        return out

    async def _list_rulings(self, client: httpx.AsyncClient, type_code: str) -> list[str]:
        # Paginate the listing page
        listing_base = f"https://www.ato.gov.au/law/view/listing.htm?docType={type_code}"
        urls: list[str] = []
        page = 1
        while True:
            resp = await client.get(f"{listing_base}&page={page}")
            if resp.status_code != 200:
                break
            tree = HTMLParser(resp.text)
            new = [a.attrs.get("href") for a in tree.css("a.document-link[href]")]
            new = [u for u in new if u and "/law/view.htm" in u]
            if not new:
                break
            urls.extend(new)
            page += 1
            if page > 50:  # safety cap
                break
        # Normalise to absolute
        return [u if u.startswith("http") else f"https://www.ato.gov.au{u}" for u in urls]

    async def _fetch_page(self, client: httpx.AsyncClient, url: str) -> str:
        try:
            resp = await client.get(url, follow_redirects=True)
            return resp.text if resp.status_code == 200 else ""
        except httpx.HTTPError:
            return ""

    def _parse_ruling(self, html: str, url: str, type_code: str, doc_type: str, out: SourceOutput) -> None:
        tree = HTMLParser(html)
        title_node = tree.css_first("h1")
        title = title_node.text(strip=True) if title_node else url
        # Ruling ID often in title like "TR 2024/5"

        if not ruling_id_match:
            return

        doc_id = f"ato-law:{type_code}/{ruling_id}"

        effective_from = None

        # Body
        cleaned = clean_html(html)
        chunks = chunk_html(cleaned, max_chars=2400, overlap_chars=200)
        if not chunks:
            return
        out.docs.append(Doc(
            doc_id=doc_id,
            source="ato",
            url=url,
            title=title,
            doc_type=doc_type,

            retrieved_at=datetime.now(timezone.utc).isoformat(),

                char_end=c.char_end,
```

```python
# sources/reclassify.py
import re
from ..schema import Doc


_URL_TO_TYPE = [
    (re.compile(r"/law/view\.htm.*docid=TXR/"), "ATO_RULING_TR"),
    (re.compile(r"/law/view\.htm.*docid=TXD/"), "ATO_RULING_TD"),
    (re.compile(r"/law/view\.htm.*docid=GSTR/"), "ATO_RULING_GSTR"),
    (re.compile(r"/law/view\.htm.*docid=GSTD/"), "ATO_RULING_GSTD"),
    (re.compile(r"/law/view\.htm.*docid=PR/"), "ATO_RULING_PR"),
    (re.compile(r"/law/view\.htm.*docid=CR/"), "ATO_RULING_CR"),
    (re.compile(r"/law/view\.htm.*docid=LCR/"), "ATO_RULING_LCR"),
    (re.compile(r"/law/view\.htm.*docid=PCG/"), "ATO_RULING_PCG"),
]


def reclassify(docs: list[Doc]) -> list[Doc]:
    out = []
    for d in docs:
        new_type = d.doc_type
        for pattern, target_type in _URL_TO_TYPE:
            if pattern.search(str(d.url)):
                new_type = target_type
                break
        if new_type != d.doc_type:
            d = d.model_copy(update={"doc_type": new_type})
        out.append(d)
    return out
```

```python
import hashlib, json, time
from pathlib import Path
import pyzstd


def build_manifest(db_path: Path, embedding_model: str) -> dict:
    sha = hashlib.sha256()
    size = 0
    with db_path.open("rb") as f:
        while True:
            chunk = f.read(1024 * 1024)
            if not chunk:
                break
            sha.update(chunk)
            size += len(chunk)
    return {
        "schema_version": "0.2.0",
        "generated_at": time.strftime("%Y-%m-%dT%H:%M:%SZ", time.gmtime()),
        "embedding_model": embedding_model,
        "embedding_dim": 384,
        "corpus_sha256": sha.hexdigest(),
        "uncompressed_size": size,
    }


def compress_corpus(db_path: Path, out_path: Path) -> int:
    with db_path.open("rb") as src, out_path.open("wb") as dst:
        pyzstd.compress_stream(src, dst, level_or_option=19)
    return out_path.stat().st_size
```
